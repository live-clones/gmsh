# API regressions use the library built alongside this executable.
if(HAVE_MESH AND TARGET shared)
  add_executable(mesh_core_api api.cpp)
  # Keep the executable beside the shared library (also needed for Windows DLLs).
  set_target_properties(mesh_core_api PROPERTIES
                        RUNTIME_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR})
  target_include_directories(mesh_core_api PRIVATE ${CMAKE_SOURCE_DIR}/api)
  target_link_libraries(mesh_core_api shared)

  set(MESH_CORE_CASES mesh_caches)
  if(HAVE_OCC)
    list(APPEND MESH_CORE_CASES occ_classification)
  endif()
  if(HAVE_SOLVER)
    list(APPEND MESH_CORE_CASES discrete_projection)
  endif()
  if(HAVE_POST AND HAVE_PLUGINS)
    target_sources(mesh_core_api PRIVATE testCoreExtendField.cpp)
    target_compile_definitions(mesh_core_api PRIVATE MESH_CORE_EXTEND_FIELD_TEST)
    list(APPEND MESH_CORE_CASES extend_field_first_evaluation)
  endif()
  foreach(CASE ${MESH_CORE_CASES})
    add_test(NAME core/${CASE} COMMAND mesh_core_api ${CASE})
    set_tests_properties(core/${CASE} PROPERTIES LABELS core TIMEOUT 30)
  endforeach()
endif()
